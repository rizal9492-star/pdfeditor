// Konfigurasi Worker PDF.js pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js'; const { PDFDocument, rgb } = PDFLib; // Simpan objek File asli, BUKAN ArrayBuffer-nya let currentFile = null; let currentPdfDoc = null; let totalPages = 0; let selectedPages = new Set(); let currentMode = 'select'; async function handleFileSelect(event) { const file = event.target.files[0]; if (!file || file.type !== 'application/pdf') return; currentFile = file; // Simpan file fisik // Buat buffer khusus HANYA untuk pdf.js (Thumbnail) const bufferForPreview = await file.arrayBuffer(); const loadingTask = pdfjsLib.getDocument({ data: bufferForPreview }); currentPdfDoc = await loadingTask.promise; totalPages = currentPdfDoc.numPages; document.getElementById('drop-area').style.display = 'none'; const grid = document.getElementById('thumbnail-grid'); grid.innerHTML = ''; selectedPages.clear(); selectedPages.add(1); if (totalPages >= 2) selectedPages.add(2); for (let pageNum = 1; pageNum <= totalPages; pageNum++) { createPageCard(pageNum, grid); } syncInputWithSelection(); updateSummary(); document.getElementById('btn-action').disabled = false; } // ... (Fungsi createPageCard, togglePageSelection, syncInputWithSelection, setMode, dan updateSummary TETAP SAMA seperti sebelumnya) ... async function processSplit() { const count = selectedPages.size; if (count === 0) return; const btn = document.getElementById('btn-action'); const btnText = document.getElementById('btn-text'); btn.disabled = true; btnText.innerHTML = ' Memproses...'; try { // PERBAIKAN BUG: Ambil ArrayBuffer BARU dari file fisik setiap kali memproses const freshBuffer = await currentFile.arrayBuffer(); const srcPdf = await PDFDocument.load(freshBuffer); const isMerge = document.getElementById('chk-merge').checked; const sortedPages = Array.from(selectedPages).sort((a, b) => a - b); if (isMerge) { const newPdf = await PDFDocument.create(); const zeroBasedIndices = sortedPages.map(p => p - 1); const copiedPages = await newPdf.copyPages(srcPdf, zeroBasedIndices); // Contoh fitur PRO tambahan: Menambahkan teks pada halaman yang dipisah copiedPages.forEach(p => { // Uncomment baris di bawah jika ingin menambahkan teks otomatis! // p.drawText('Diedit dengan PDF Pro', { x: 50, y: 50, size: 12, color: rgb(1, 0, 0) }); newPdf.addPage(p); }); const pdfBytes = await newPdf.save(); downloadBlob(new Blob([pdfBytes], { type: 'application/pdf' }), `split_merged_${currentFile.name}`); } else { const zip = new JSZip(); for (const pageNum of sortedPages) { const singlePdf = await PDFDocument.create(); const [copiedPage] = await singlePdf.copyPages(srcPdf, [pageNum - 1]); singlePdf.addPage(copiedPage); const singlePdfBytes = await singlePdf.save(); zip.file(`halaman_${pageNum}.pdf`, singlePdfBytes); } const zipContent = await zip.generateAsync({ type: 'blob' }); downloadBlob(zipContent, `hasil_pisah.zip`); } } catch (err) { alert('Terjadi kesalahan: ' + err.message); } finally { btn.disabled = false; btnText.textContent = 'Pisahkan PDF ➔'; } } function downloadBlob(blob, filename) { const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = filename; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); } // Tambahkan sisa fungsi UI di sini (seperti updateCardUI, dll dari kode sebelumnya)
